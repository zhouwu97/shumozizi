% !TEX program = xelatex
% =============================================================================
%  全国大学生数学建模竞赛（CUMCM）LaTeX 论文模板 —— 官方 cumcmthesis 类
%
%  设计目标：采用官方《全国大学生数学建模竞赛论文格式规范》对应的 cumcmthesis
%  LaTeX 类，可靠、干净、可复用。
%    - 官方类提供承诺书/编号页（\maketitle）、摘要环境（abstract + \keywords）、
%      中文节编号（一、二、三、），默认 A4 2.5cm 边距、1.38 行距。
%    - 本模板覆盖其默认字体 STSong/STKaiti 为 SimSun/SimHei/KaiTi/FangSong，
%      缺失时回退 Fandol（TeX 发行版自带）。
%    - 缺少正文 sections/questions.tex 时明确报错，不静默产出空白论文。
%    - 缺少可选内容（参考文献 / 附录）时不产生空白页或伪造内容。
%
%  编译方式：xelatex main.tex（含交叉引用时编译两遍）。
%
%  与工作流的关系：本文件是「正式模板」。可编译的演示样例在
%    skills/5writing/examples/cumcm-xelatex/（不随模板实例化进入每个 run）。
%    模板实例化时会自动生成 sections/questions.tex 文件（题目数决定的正文骨架），
%    因此本目录的 sections/ 不应预置 questions.tex。
% =============================================================================
\documentclass[withoutpreface,bwprint,12pt]{cumcmthesis}

% =========================================================================
%  论文配置块（写作阶段按本题填写 / 切换开关）
% =========================================================================
% 标题：\title 供给 \maketitle 的标题页；题号/队号/学校/成员等在提交时填写。
\newcommand{\PaperTitle}{论文标题}
\title{\PaperTitle}
\tihao{}                    % 题号（如 C）
\baominghao{}               % 队号（提交时填写）
% \schoolname{}{} \yearinput{} \monthinput{} \dayinput{} 等按需填写。
% 是否包含参考文献。为真时缺 references.tex 会明确报错，而非静默留空。
\newif\ifpaperhasreferences   \paperhasreferencestrue
% 参考文献标题是否参与章节编号（true → “六、参考文献”；false → 无编号）。
\newif\ifnumberedreferences   \numberedreferencesfalse
% 是否输出附录。为假时完全不产生附录标题与附录页。
\newif\ifshowappendix         \showappendixtrue

% =========================================================================
%  正文排版
% =========================================================================
% 国赛正文按宋体小四（12pt）排版；cumcmthesis 已设 2.5cm 边距与 1.38 行距。
% 显式锁定 12pt/18pt（小四、1.5 倍行距）以满足官方排版契约。
\usepackage{setspace}
\fontsize{12pt}{18pt}\selectfont
\setlength{\parindent}{2em}
\setlength{\parskip}{0pt}
\raggedbottom

% ---------- 页码：摘要首页即第 1 页；无页眉，底部居中（article plain 默认）----------
% cumcmthesis \maketitle 会把页码重置为 1，摘要页用 \thispagestyle{plain} 显示页码。

% ---------- 数学：变量使用 Times 系斜体 ----------
\IfFileExists{newtxmath.sty}{\usepackage{newtxmath}}{\usepackage{amssymb,mathptmx}}
\allowdisplaybreaks[2]

% ---------- 图表 ----------
\usepackage{graphicx}
% 兼容独立编译（figures/）与实例化 run 布局（../figures/current/）。
\graphicspath{{figures/}{figures/current/}{../figures/current/}}
\usepackage{booktabs,tabularx,longtable,array,multirow}
\usepackage{float}
% placeins 不用 [section]：不自动在每节末尾拦截浮动体，交由手写 \FloatBarrier 控制。
\usepackage{placeins}
\usepackage{caption}
\DeclareCaptionFont{cncaption}{\fontsize{10.5pt}{14pt}\selectfont}
\captionsetup{

  justification=centering,

% ---------- 黑白超链接 ----------
\usepackage[hidelinks]{hyperref}

% ---------- 参考文献标题：单一来源，避免双标题 ----------
% thebibliography 默认自行输出 \section*{参考文献}（不编号）。
% 若 \ifnumberedreferences 为真，则把它就地改成编号的 \section{参考文献}。
\usepackage{etoolbox}
\AtBeginDocument{%
  \ifnumberedreferences
    \patchcmd{\thebibliography}{\section*{\refname}}{\section{\refname}}{}{%
      \PackageWarning{cumcm-template}{无法为参考文献标题加编号：thebibliography 结构与预期不符}%
    }%
  \fi
}

% 数学宏（正文使用）。
\newcommand{\logit}[1]{\mathrm{logit}\left(#1\right)}
\newcommand{\R}{\mathbb{R}}

% 直接答案框：正文用 \begin{answerbox}...\end{answerbox} 展示直接答案。
\newenvironment{answerbox}{%
  \begin{center}\begin{minipage}{0.96\textwidth}\noindent\rule{\linewidth}{0.5pt}\\[2pt]
  \noindent{\heiti\fontsize{14pt}{18pt}\selectfont 直接答案：}}%
  {\par\vspace{2pt}\noindent\rule{\linewidth}{0.5pt}\end{minipage}\end{center}}

% \abstractcn{摘要正文}{关键词}：使用 cumcmthesis 官方摘要环境。
% 标题页 \maketitle 已在文档开头调用；本命令在摘要页显示页码 1 并输出摘要与关键词。
\newcommand{\abstractcn}[2]{%
  \thispagestyle{plain}%
  \begin{abstract}%
    #1\par
    \keywords{#2}%
  \end{abstract}%
}

% \threelinetable{表题}{列定义}{表头}{表体}：三线表。
\newcommand{\threelinetable}[5][]{%
  \begin{table}[htbp]
    \centering
    \caption{#2}%
    \if\relax\detokenize{#1}\relax\else\label{#1}\fi
    \begin{tabular}{#3}
      \toprule
      #4 \\
      \midrule
      #5 \\
      \bottomrule
    \end{tabular}
  \end{table}
}

% \appendixsection{标题}：附录小节，自动编号“附录 A / 附录 B / …”。
\newcounter{paperappendix}
\newcommand{\appendixsection}[1]{%
  \stepcounter{paperappendix}%
  \phantomsection%
  \section*{附录~\Alph{paperappendix}\quad #1}%
}

% \referencescn：参考文献节。references.tex 用 thebibliography 环境放条目。
\newcommand{\referencescn}{%
  \ifpaperhasreferences
    \FloatBarrier
    \IfFileExists{references.tex}{%
      \begingroup
        \setstretch{1.18}%
        \fontsize{10.5pt}{14pt}\selectfont
        \input{references.tex}%
      \endgroup
    }{%
      \PackageError{cumcm-template}%
        {references.tex is required when paperhasreferences is true}%
        {Provide references.tex, or set \noexpand\paperhasreferencesfalse in the config block.}%
    }%
  \fi
}

% \appendixcn：默认附录（五节）。按国赛惯例展开为：
%   A 附件与数据说明 / B 运行环境与可复现性 / C 关键结果表 /
%   D 模型关键公式 / E 核心源代码。
\newcommand{\appendixcn}{%
  \FloatBarrier
  \clearpage
  \IfFileExists{sections/appendices/attachments.tex}{%
    \appendixsection{附件与数据说明}\input{sections/appendices/attachments}%
  }{}%
  \IfFileExists{sections/appendices/reproducibility.tex}{%
    \appendixsection{运行环境与可复现性}\input{sections/appendices/reproducibility}%
  }{}%
  \IfFileExists{sections/appendices/key_results.tex}{%
    \appendixsection{关键结果表}\input{sections/appendices/key_results}%
  }{}%
  \IfFileExists{sections/appendices/key_formulas.tex}{%
    \appendixsection{模型关键公式}\input{sections/appendices/key_formulas}%
  }{}%
  \IfFileExists{sections/A_code.tex}{%
    \appendixsection{核心源代码}\input{sections/A_code}%
  }{}%
}

\begin{document}
% cumcmthesis \maketitle（withoutpreface）：标题页 + 页码重置为 1。
\maketitle

% 摘要即第 1 页（含页码），随后正文。
\abstractcn{%
  [背景：核心困难是什么，使这道题非显然]。本文将[问题的核心困难]定义为[建模结构]，
  以[主方法]求解。得到[最重要数值结果，含具体数字，如：最优策略总收益为 X，
  较单动作基线提升 Y\%]。分析发现[核心规律，如：存在边际收益递减，
  瓶颈来自[约束名称]]，据此[决策含义]。模型在[参数范围]内保持稳定；
  当[关键假设改变]时结论可能随之调整。%
}{%
  关键词一\quad 关键词二\quad 关键词三\quad 关键词四%
}

% 正文入口：优先读取实例化生成的 sections/questions.tex 文件；缺失时明确报错。
% 正文不设目录。
\IfFileExists{sections/questions.tex}{%
  \input{sections/questions.tex}%
}{%
  \PackageError{cumcm-template}%
    {sections/questions.tex is required (paper body missing)}%
    {Create sections/questions.tex with the paper body, then recompile.}%
}

\referencescn

\ifshowappendix
  \appendixcn
\fi

\end{document}
