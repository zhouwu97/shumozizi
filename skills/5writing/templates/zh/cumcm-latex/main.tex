% ---------- 参考文献标题：单一来源，避免双标题 ----------
% thebibliography 默认自行输出 \section*{参考文献}（不编号）。
% 若 \ifnumberedreferences 为真，则把它就地改成编号的 \section{参考文献}。
% 无论哪种情形，“参考文献”标题都只此一处，\referencescn 不再另行输出标题。
\usepackage{etoolbox}
\AtBeginDocument{%
  \ifnumberedreferences
    \patchcmd{\thebibliography}{\section*{\refname}}{\section{\refname}}{}{%
      \PackageWarning{cumcm-template}{无法为参考文献标题加编号：thebibliography 结构与预期不符}%
    }%
  \fi
}

% \papertitle{标题}：首页居中大标题（首页不是独立封面）。
\newcommand{\papertitle}[1]{%
  \vspace*{0.55cm}%
  \begin{center}
    {\heiti\fontsize{18pt}{24pt}\selectfont #1\par}
  \end{center}
  \vspace{0.65cm}%
}

% \abstractcn{摘要正文}{关键词}：居中“摘要”标题 + 正文 + 加粗“关键词：” + 换页。
% 摘要建议按叙述逻辑组织（背景与困难 → 建模结构 → 主方法 → 关键数值结果 →
% 规律与决策含义 → 可信边界），不要机械按“问题一/二/三”逐句点名。
\newcommand{\abstractcn}[2]{%
  \begin{center}
    {\heiti\fontsize{15pt}{20pt}\selectfont 摘要\par}
  \end{center}
  \vspace{0.35em}%
  \begingroup
    \setstretch{1.32}%
    \fontsize{12pt}{18pt}\selectfont
    #1\par
    \vspace{0.65em}%
    \noindent{\heiti 关键词：}#2\par
  \endgroup
  \clearpage
}

% \threelinetable{表题}{列定义}{表头}{表体}：三线表。
% 表题只写内容，不要手写“表 1”；需要交叉引用时优先直接用 table 环境并加 \label。
% 使用 [htbp] 让浮动体正常参与排版，避免 [H] 造成的大片空白与越界。
% 可选参数为交叉引用标签：\threelinetable[tab:key]{标题}{列格式}{表头}{表体}。
% 省略 [tab:key] 时不产生标签，向后兼容旧调用。
\newcommand{\threelinetable}[5][]{%
  \begin{table}[htbp]
    \centering
    \caption{#2}%
    \if\relax\detokenize{#1}\relax\else\label{#1}\fi
    \begin{tabular}{#3}
      \toprule
      #4 \\
      \midrule
      #5 \\
      \bottomrule
    \end{tabular}
  \end{table}
}

% \appendixsection{标题}：附录小节，自动编号“附录 A / 附录 B / …”。
% 支持单附录与多附录：单附录调用一次，多附录多次调用即可。不自动 \appendix。
\newcounter{paperappendix}
\newcommand{\appendixsection}[1]{%
  \stepcounter{paperappendix}%
  \phantomsection%
  \section*{附录~\Alph{paperappendix}\quad #1}%
}

% \referencescn：参考文献节。references.tex 用 thebibliography 环境放条目。
% 关键：标题“参考文献”只由 thebibliography 自动输出一次，本命令**不再手写标题**，
% 否则会与 thebibliography 的内置标题叠加出现两个“参考文献”。
% 是否编号由 \ifnumberedreferences 控制（见下方 \AtBeginDocument 补丁）。
% 缺 references.tex 且 \ifpaperhasreferences 为真时明确报错（不伪造条目）。
\newcommand{\referencescn}{%
  \ifpaperhasreferences
    \FloatBarrier
    \IfFileExists{references.tex}{%
      \begingroup
        \setstretch{1.18}%
        \fontsize{10.5pt}{14pt}\selectfont
        \input{references.tex}%
      \endgroup
    }{%
      \PackageError{cumcm-template}%
        {references.tex is required when paperhasreferences is true}%
        {Provide references.tex, or set \noexpand\paperhasreferencesfalse in the config block.}%
    }%
  \fi
}

% \appendixcn：默认附录（核心代码）。sections/A_code.tex 不存在时不产生任何附录页。
\newcommand{\appendixcn}{%
  \FloatBarrier
  \IfFileExists{sections/A_code.tex}{%
    \clearpage
    \appendixsection{核心代码}%
    \input{sections/A_code}%
  }{}%
}
